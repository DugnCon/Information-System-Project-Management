\chapter*{Mở đầu}
\addcontentsline{toc}{chapter}{Mở đầu}

\textbf{Lý do chọn đề tài.} Vượt chi phí và trễ tiến độ là hiện tượng phổ biến ở các dự án xây dựng, nhất là các dự án vốn lớn, nhiều bên tham gia như trong ngành dầu khí. Quản lý rủi ro là một trong mười vùng kiến thức của PMBOK~\cite{pmbok6} và là nội dung trọng tâm của môn Quản lý dự án hệ thống thông tin. Bài báo của Kassem~\cite{kassem2022} là một nghiên cứu điển hình kết hợp các công cụ quản lý rủi ro truyền thống (chỉ số RII, ma trận xác suất -- tác động) với một phương pháp thống kê hiện đại (mô hình phương trình cấu trúc PLS-SEM). Vì vậy bài báo phù hợp để vừa học kiến thức nền, vừa thực hành phân tích dữ liệu.

\textbf{Bài báo được nghiên cứu.} Kassem, M.A. (2022). \emph{Risk Management Assessment in Oil and Gas Construction Projects Using Structural Equation Modeling (PLS-SEM)}. Tạp chí \emph{Gases}, tập 2, số 2, trang 33--60. Bài báo khảo sát 314 người tham gia các dự án xây dựng dầu khí ở Yemen, nhận diện 51 yếu tố rủi ro thuộc 13 nhóm, và xây dựng mô hình đo ảnh hưởng của rủi ro nội bộ và rủi ro bên ngoài tới thành công của dự án.

\textbf{Mục tiêu của báo cáo.}
\begin{enumerate}
  \item Trình bày các khái niệm nền tảng về rủi ro và quản lý rủi ro dự án, cùng các phương pháp mà bài báo sử dụng: thành phần, cách hoạt động, ưu điểm, hạn chế và xu hướng.
  \item Tái hiện phần phân tích dữ liệu của bài báo. Vì dữ liệu gốc không được công bố, việc tái hiện theo hai hướng: kiểm tra lại số liệu đã công bố, và chạy lại toàn bộ quy trình trên dữ liệu mô phỏng.
  \item Phân tích hai ví dụ thực tế, gồm một dự án của ngành dầu khí Việt Nam và một dự án hệ thống thông tin, đồng thời minh họa cách dùng các công cụ MS Project, Excel và mô phỏng Monte Carlo.
  \item Đánh giá phản biện bài báo và đề xuất cải thiện.
\end{enumerate}

\textbf{Bố cục.} Ngoài phần mở đầu và kết luận, báo cáo gồm bốn chương:
\begin{itemize}
  \item Chương~\ref{chuong:tong-quan} trình bày tổng quan về quản lý rủi ro dự án.
  \item Chương~\ref{chuong:phuong-phap} trình bày các phương pháp RII, PIM, PLS-SEM và mô hình nghiên cứu của bài báo.
  \item Chương~\ref{chuong:tai-hien} trình bày kết quả tái hiện thực nghiệm.
  \item Chương~\ref{chuong:mo-rong} phân tích hai ví dụ thực tế, mô phỏng Monte Carlo và đánh giá phản biện.
\end{itemize}
